# new generic Makefile -- compiles from ./src to ./build

CC = gcc
CFLAGS = -g -Wall
LDFLAGS =
RM = rm -v
BUILD = ./build
SRC = ./src
LIBS = ./libs
TESTS = ./testing

SRCS = $(wildcard $(SRC)/*.c)
PROGS = $(patsubst $(SRC)/%.c,$(BUILD)/%,$(SRCS))

all: newbuilddir $(PROGS)

.PHONY: newbuilddir
newbuilddir:
	@mkdir -p $(BUILD)

$(PROGS): $(SRCS)
	$(CC) $(CFLAGS) -o $@ $< $(LDFLAGS)

.PHONY: clean
clean:
	$(RM) $(PROGS)

.PHONY: dirs
dirs:
	@mkdir -pv $(BUILD) $(SRC) $(LIBS) $(TESTS)

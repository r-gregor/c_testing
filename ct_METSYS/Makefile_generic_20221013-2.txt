# new generic Makefile -- compiles from ./src to ./build

CC = gcc
CFLAGS = -g -Wall
LDFLAGS =
RM = rm -v
BUILD = ./build
SRC = ./src
LIBS = ./libs
INCL = ./include
TESTS = ./testing

SRCS = $(wildcard $(SRC)/*.c)
PROGS = $(patsubst $(SRC)/%.c,$(BUILD)/%,$(SRCS))

all: newbuilddir $(PROGS)

.PHONY: newbuilddir
newbuilddir:
	@mkdir -p $(BUILD)

$(PROGS): $(SRCS)
	@echo -n "Compiling $@ ... "
	@$(CC) $(CFLAGS) -o $@ $< $(LDFLAGS)
	@echo done

.PHONY: clean
clean:
	@echo "Removing executables ..."
	@$(RM) $(PROGS)

.PHONY: dirs
dirs:
	@echo "Creating dirs ..."
	@mkdir -pv $(BUILD) $(SRC) $(LIBS) $(INCL) $(TESTS)
